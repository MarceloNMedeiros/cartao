// vCard data
const vCardData = `BEGIN:VCARD
VERSION:3.0
N:Medeiros;Marcelo;Neves de;;PhD
FN:Marcelo Neves de Medeiros, PhD
ORG:INMETRO - Instituto Nacional de Metrologia, Qualidade e Tecnologia

TEL;TYPE=CELL:+5521994974704
EMAIL;TYPE=WORK:mnmedeiros@inmetro.gov.br
URL:https://www.gov.br/inmetro
END:VCARD`;

// Generate QR Code
document.addEventListener('DOMContentLoaded', function() {
  new QRCode(document.getElementById("qrcode"), {
    text: vCardData,
    width: 160,
    height: 160,
    colorDark: "#004079",
    colorLight: "#ffffff",
    correctLevel: QRCode.CorrectLevel.M
  });
});

// Download vCard
function downloadVCard() {
  const blob = new Blob([vCardData], { type: 'text/vcard;charset=utf-8' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = 'Marcelo_Medeiros_Inmetro.vcf';
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
  showToast('Contato baixado com sucesso!');
}

// Share via Web Share API or fallback
async function shareCard() {
  const shareData = {
    title: 'Marcelo Neves de Medeiros, PhD',
    text: 'Pesquisador Tecnologista em Metrologia e Qualidade — INMETRO',
    url: window.location.href
  };

  if (navigator.share) {
    try {
      await navigator.share(shareData);
    } catch (err) {
      if (err.name !== 'AbortError') {
        fallbackCopy();
      }
    }
  } else {
    fallbackCopy();
  }
}

function fallbackCopy() {
  const text = `Marcelo Neves de Medeiros, PhD
Pesquisador Tecnologista em Metrologia e Qualidade
INMETRO
Tel: +55 (21) 2145-3321
WhatsApp: +55 (21) 99497-4704
E-mail: mnmedeiros@inmetro.gov.br
Site: inmetro.gov.br`;

  navigator.clipboard.writeText(text).then(() => {
    showToast('Dados copiados para a área de transferência!');
  }).catch(() => {
    showToast('Não foi possível copiar.');
  });
}

function showToast(msg) {
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.classList.add('show');
  setTimeout(() => t.classList.remove('show'), 2500);
}
